1"""identity/ tools - who is calling. 

2 

3Owner: the identity lane. Replace each ``stub_*`` call with the real logic. 

4Keep the signatures exactly as ``vortex/contract.py`` declares them. 

5 

6What the docs say this lane must get right (clinic docs, "Six things worth knowing"): 

7 

8- /directory filters on exact fields. A field that does not match excludes the 

9 patient. Use name + date_of_birth to split two people with the same name. 

10- Confirm on a second field before you trust a match. 

11- Submit the record's name and ids, never what the caller said. 

12- ``from_number`` is a hint, never identification. The caller is not always 

13 the patient (a parent for a child, a daughter for her father). 

14- A caller the directory does not know cannot be booked. Register first. 

15- DNI: 8 digits + letter. NIE: X/Y/Z + 7 digits + letter. The letter is 

16 ``"TRWAGMYFPDXBNJZSQVHLCKE"[number % 23]`` where a NIE's leading X/Y/Z 

17 counts as 0/1/2. A wrong letter is a 422 at submit time. 

18"""

39_CHECK_LETTERS = "TRWAGMYFPDXBNJZSQVHLCKE" 

40_NIE_PREFIX_DIGIT = {"X": "0", "Y": "1", "Z": "2"} 

41_DNI_RE = re.compile(r"\d{8}[A-Z]") 

42_NIE_RE = re.compile(r"[XYZ]\d{7}[A-Z]")

48def _unanimous(values: list[str]) -> str | None: 

49 """The one value every item shares, if there are enough of them to call it a pattern.""" 

50 return values[0] if len(values) >= _MIN_SUPPORT and len(set(values)) == 1 else None

53def _mine_preferences(history: list[Appointment]) -> dict[str, Any]: 

54 """Patterns in past visits: where/who they always went to, per appointment type too. 

55 

56 Not a rejection or a rule — a set of soft defaults the conversation can offer 

57 ("you're usually at Sur, want that again?") instead of asking cold. 

58 """ 

59 locations = [a.location_id for a in history] 

60 providers = [a.provider_id for a in history] 

61 by_type: dict[str, list[Appointment]] = {} 

62 for a in history: 

63 by_type.setdefault(a.appointment_type_id, []).append(a) 

64 

65 return { 

66 "visit_count": len(history), 

67 "location_preference": _unanimous(locations), 

68 "distinct_locations": sorted(set(locations)), 

69 "provider_preference": _unanimous(providers), 

70 "distinct_providers": sorted(set(providers)), 

71 "type_provider_preference": { 

72 t: p 

73 for t, items in by_type.items() 

74 if (p := _unanimous([a.provider_id for a in items])) 

75 }, 

76 "type_location_preference": { 

77 t: loc 

78 for t, items in by_type.items() 

79 if (loc := _unanimous([a.location_id for a in items])) 

80 }, 

81 }

84async def _postprocess_patient(ctx: ToolContext, patient: PatientRecord) -> dict[str, Any]: 

85 """Background work kicked off the moment a caller is identified. 

86 

87 Never raises: nothing on the happy path ever awaits this task, so an 

88 unhandled exception would otherwise only surface as an asyncio "exception 

89 never retrieved" warning, silently. 

90 """ 

91 try: 

92 history = await ctx.clinic.appointments(patient.patient_id, when="past") 

93 return _mine_preferences(history) 

94 except Exception as exc: 

95 ctx.log.event("patient_postprocess.failed", detail=f"{type(exc).__name__}: {exc}") 

96 return {}

99async def find_patient(ctx: ToolContext, args: FindPatientInput) -> FindPatientResult: 

100 """Look the caller up in /directory and decide: found, ambiguous or not_found.""" 

101 phone = args.phone 

102 gave_nothing = not any((args.name, args.national_id, args.phone, args.date_of_birth)) 

103 if gave_nothing: 

104 # Caller gave nothing else to search on: fall back to the line's own number. 

105 phone = ctx.from_number 

106 

107 candidates = await ctx.clinic.directory( 

108 name=args.name, national_id=args.national_id, phone=phone, date_of_birth=args.date_of_birth 

109 ) 

110 if not candidates: 

111 return FindPatientResult(status="not_found") 

112 if len(candidates) == 1: 

113 patient = candidates[0] 

114 # Fire-and-forget: mine their visit history while the conversation carries 

115 # on, so a preference is already there by the time it's needed. 

116 ctx.state[PATIENT_POSTPROCESS_KEY] = asyncio.create_task( 

117 _postprocess_patient(ctx, patient) 

118 ) 

119 return FindPatientResult(status="found", patient=patient) 

120 

121 # Ambiguous: ask for whichever unfilled field would split the candidates. 

122 # date_of_birth is the go-to disambiguator; name is last since it's already given. 

123 for field_name, given in ( 

124 ("date_of_birth", args.date_of_birth), 

125 ("national_id", args.national_id), 

126 ("phone", args.phone), 

127 ("name", args.name), 

128 ): 

129 if given is None: 

130 return FindPatientResult(status="ambiguous", candidates=candidates, ask_for=field_name) 

131 return FindPatientResult(status="ambiguous", candidates=candidates)

134async def validate_national_id(ctx: ToolContext, args: ValidateNationalIdInput) -> NationalIdCheck: 

135 """Normalise a spoken DNI/NIE and check its letter (mod-23).""" 

136 normalized = args.value.strip().upper().replace(" ", "").replace("-", "") 

137 if _DNI_RE.fullmatch(normalized): 

138 digits, letter, kind = normalized[:8], normalized[8], "dni" 

139 elif _NIE_RE.fullmatch(normalized): 

140 digits = _NIE_PREFIX_DIGIT[normalized[0]] + normalized[1:8] 

141 letter, kind = normalized[8], "nie" 

142 else: 

143 return NationalIdCheck(normalized=normalized, kind="invalid", valid=False) 

144 expected = _CHECK_LETTERS[int(digits) % 23] 

145 return NationalIdCheck( 

146 normalized=normalized, kind=kind, valid=letter == expected, expected_letter=expected 

147 )

150def _normalize_phone(phone: str) -> str: 

151 """As dictated -> E.164. A bare 9-digit Spanish number gets the +34 prefix.""" 

152 had_plus = phone.strip().startswith("+") 

153 digits = "".join(ch for ch in phone if ch.isdigit()) 

154 if had_plus: 

155 return f"+{digits}" 

156 if len(digits) == 9: 

157 return f"+34{digits}" 

158 return digits

161async def build_registration(ctx: ToolContext, args: BuildRegistrationInput) -> RegistrationResult: 

162 """Turn dictated demographics into a ``RegisterAction``. 

163 

164 The national id's check letter is not re-validated here: the conversation 

165 is expected to have already confirmed it with ``validate_national_id`` 

166 (asking the caller to repeat a wrong digit) before demographics are 

167 complete enough to call this. A malformed id still 422s at submit time. 

168 """ 

169 check = await validate_national_id(ctx, ValidateNationalIdInput(value=args.national_id)) 

170 return RegistrationResult( 

171 action=RegisterAction( 

172 given_name=args.given_name, 

173 first_surname=args.first_surname, 

174 second_surname=args.second_surname, 

175 national_id=check.normalized, 

176 date_of_birth=args.date_of_birth, 

177 phone=_normalize_phone(args.phone), 

178 email=args.email.strip().lower(), 

179 insurer=args.insurer, 

180 ) 

181 )
