1"""Turning a spoken street address into the nearest site that can serve it. 

2 

3Problem 15 gives a real Madrid-area address and no site name. Ground truth is 

4the smallest straight-line distance to the coordinates the clinic publishes, 

5among the sites that can actually serve the request — and winners win by a 

6clear margin, so the address only has to land in the right part of the city. 

7 

8Two layers, in order: 

9 

101. **A gazetteer**, below: the municipalities of the Madrid area and the city's 

11 main streets and districts. No network, deterministic, and enough for a 

12 margin measured in kilometres. 

132. **A live geocoder**, when ``Settings.geocoder_url`` (``VORTEX_GEOCODER_URL``) 

14 names a Nominatim-compatible endpoint. Off by default so that evals and 

15 offline work never depend on a network call, and so nothing leaks a 

16 caller's address to a third party unless the team turned it on 

17 deliberately. 

18 

19If neither places the address, the site whose own published address shares the 

20most words with it answers. A caller who gives an address we cannot place is 

21not a refusal: it is a question to ask them. 

22"""

39def fold(text: str) -> str: 

40 """Lower-cased, accent-folded, punctuation-free, single-spaced.""" 

41 stripped = unicodedata.normalize("NFKD", text) 

42 stripped = "".join(ch for ch in stripped if not unicodedata.combining(ch)) 

43 return re.sub(r"\s+", " ", re.sub(r"[^a-z0-9 ]+", " ", stripped.lower())).strip()

46def haversine_km(a: tuple[float, float], b: tuple[float, float]) -> float: 

47 """Straight-line distance between two (lat, lon) pairs, in kilometres.""" 

48 lat1, lon1 = math.radians(a[0]), math.radians(a[1]) 

49 lat2, lon2 = math.radians(b[0]), math.radians(b[1]) 

50 dlat, dlon = lat2 - lat1, lon2 - lon1 

51 h = math.sin(dlat / 2) ** 2 + math.cos(lat1) * math.cos(lat2) * math.sin(dlon / 2) ** 2 

52 return 2 * EARTH_RADIUS_KM * math.asin(math.sqrt(h))

190def gazetteer_lookup(address: str) -> tuple[float, float] | None: 

191 """The most specific gazetteer entry the address mentions.""" 

192 text = fold(address) 

193 best: tuple[int, tuple[float, float]] | None = None 

194 for name, point in GAZETTEER.items(): 

195 if re.search(rf"\b{re.escape(name)}\b", text) and (best is None or len(name) > best[0]): 

196 best = (len(name), point) 

197 return best[1] if best else None

200async def geocode_live( 

201 address: str, settings: Settings | None = None 

202) -> tuple[float, float] | None: 

203 """Ask a Nominatim-compatible endpoint. ``None`` unless one is configured.""" 

204 url = (settings or get_settings()).geocoder_url 

205 if not url: 

206 return None 

207 import httpx # local: the offline path must not need it 

208 

209 params = {"q": address, "format": "json", "limit": 1, "countrycodes": "es"} 

210 try: 

211 async with httpx.AsyncClient(timeout=GEOCODER_TIMEOUT_SECS) as http: 

212 response = await http.get( 

213 url, params=params, headers={"User-Agent": "vortex-clinic-agent/1.0"} 

214 ) 

215 if response.status_code >= 400: 

216 return None 

217 hits = response.json() 

218 except Exception: # noqa: BLE001 - a geocoder outage must not lose the call 

219 return None 

220 if not hits: 

221 return None 

222 try: 

223 return float(hits[0]["lat"]), float(hits[0]["lon"]) 

224 except (KeyError, TypeError, ValueError): 

225 return None

228@lru_cache(maxsize=512) 

229def _address_words(address: str) -> frozenset[str]: 

230 return frozenset(w for w in fold(address).split() if w not in _NOISE and not w.isdigit())

233def address_overlap(caller: str, site_address: str) -> int: 

234 """How many meaningful words two addresses share. The last resort.""" 

235 return len(_address_words(caller) & _address_words(site_address))

238async def locate(address: str, settings: Settings | None = None) -> tuple[float, float] | None: 

239 """Coordinates for a spoken address: gazetteer first, then a live geocoder.""" 

240 return gazetteer_lookup(address) or await geocode_live(address, settings)

243def nearest(point: tuple[float, float], sites: list[Any]) -> tuple[Any, float] | None: 

244 """The site closest to ``point``, with its distance. Sites need coordinates.""" 

245 placed = [s for s in sites if s.latitude is not None and s.longitude is not None] 

246 if not placed: 

247 return None 

248 best = min(placed, key=lambda s: haversine_km(point, (s.latitude, s.longitude))) 

249 return best, haversine_km(point, (best.latitude, best.longitude))
