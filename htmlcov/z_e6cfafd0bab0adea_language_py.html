1"""Language handling: detect, switch mid-call, constrain the provider. 

2 

3Owner: the conversation lane. 

4 

5The board checks the right language was used; the jury checks it was used 

6well. Every provider speaks Spanish; only four speak Catalan (problem 11). 

7 

8Soniox ``stt-rt-v5`` tags every token with the language it heard when 

9``enable_language_identification`` is on, so the ``TranscriptionFrame.language`` 

10attribute is the signal we trust. The word list below is only a fallback for 

11the frames that arrive untagged. 

12 

13TODO(conversation): 

14- Pass the language into ``find_slots(language=...)`` only when the caller 

15 asks for a doctor they can talk to; elsewhere any provider is fine. 

16- Widen the fallback markers once we hear real callers (gl, eu). 

17""" 

18 

19from __future__ import annotations 

20 

21from typing import TYPE_CHECKING, Any 

22 

23if TYPE_CHECKING: # pragma: no cover - import only for the annotation 

24 from pipecat.transcriptions.language import Language 

25 

26 from vortex.settings import Settings 

27 

28SUPPORTED_LANGUAGES: tuple[str, ...] = ("es", "ca", "en", "gl", "eu") 

29 

30DEFAULT_LANGUAGE = "es" 

31 

32# Words that are common in the language and rare in Spanish. Lowercase, no 

33# accents stripped: the STT writes them accented. 

34_MARKERS: dict[str, tuple[str, ...]] = { 

35 "ca": ("bon dia", "vull", "si us plau", "gràcies", "amb", "hora", "això", "però", "nosaltres"), 

36 "en": ("hello", "good morning", "appointment", "please", "thank you", "i would like"), 

37 "gl": ("bos días", "grazas", "quero", "moitas grazas"), 

38 "eu": ("egun on", "eskerrik asko", "mesedez", "nahi dut"), 

39}

42def normalise_language(code: object) -> str | None: 

43 """Fold any language code (``Language.ES_ES``, ``"es-ES"``, ``"es"``) to ours.""" 

44 if code is None: 

45 return None 

46 try: 

47 text = str(getattr(code, "value", code)).strip().lower() 

48 except Exception: # never let a stray object break a live call 

49 return None 

50 if not text: 

51 return None 

52 base = text.replace("_", "-").split("-", 1)[0] 

53 return base if base in SUPPORTED_LANGUAGES else None

56def detect_language(transcript: str, hint: object | None = None) -> str: 

57 """Return an ISO-639-1 code from ``SUPPORTED_LANGUAGES``. 

58 

59 ``hint`` is the per-token language Soniox reports on the transcription 

60 frame. When it is present and supported it wins; otherwise a short marker 

61 list decides, and Spanish is the default. Never raises. 

62 """ 

63 from_hint = normalise_language(hint) 

64 if from_hint: 

65 return from_hint 

66 try: 

67 text = (transcript or "").lower() 

68 except Exception: 

69 return DEFAULT_LANGUAGE 

70 if not text: 

71 return DEFAULT_LANGUAGE 

72 for code, markers in _MARKERS.items(): 

73 if any(marker in text for marker in markers): 

74 return code 

75 return DEFAULT_LANGUAGE

78def tts_voice_for( 

79 language: str, settings: Settings | Any, provider: str | None = None 

80) -> tuple[str, Language]: 

81 """The voice and the pipecat ``Language`` for a detected language. 

82 

83 Per provider, because the languages are not evenly covered: 

84 

85 - google es / ca / gl / eu (Chirp 3 HD for Spanish, Standard for the rest) 

86 - elevenlabs es 

87 

88 ``provider`` names the service the answer is for; it defaults to the 

89 primary (``settings.tts_provider``). When a primary and an alternate are 

90 both running, the caller passes the one that serves this language — 

91 ``Settings.tts_provider_for(language)`` decides which that is. 

92 

93 Anything the named provider cannot say falls back to Spanish, the one 

94 language the whole clinic speaks. Never raises: a failed lookup during a 

95 live call must not end the call. 

96 """ 

97 from pipecat.transcriptions.language import Language 

98 

99 google = { 

100 "es": (getattr(settings, "google_tts_voice_es", ""), Language.ES_ES), 

101 "ca": (getattr(settings, "google_tts_voice_ca", ""), Language.CA_ES), 

102 "gl": (getattr(settings, "google_tts_voice_gl", ""), Language.GL_ES), 

103 "eu": (getattr(settings, "google_tts_voice_eu", ""), Language.EU_ES), 

104 } 

105 # ElevenLabs takes a bare "es"; the regional code only earns a "not 

106 # verified" warning on the way through pipecat's resolver. 

107 elevenlabs = {"es": (getattr(settings, "elevenlabs_voice_id_es", ""), Language.ES)} 

108 

109 name = str(provider or getattr(settings, "tts_provider", "google") or "google").lower() 

110 voices = {"google": google, "elevenlabs": elevenlabs}.get(name, google) 

111 

112 code = normalise_language(language) or DEFAULT_LANGUAGE 

113 return voices.get(code) or voices[DEFAULT_LANGUAGE]
