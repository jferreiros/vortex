1"""The clinic's standing restrictions, derived from the catalogue. 

2 

3``/availability`` is the authority: when it refuses a provider it names the 

4rule in ``blocked``, and that name is the ``reason`` we submit. But it answers 

5about *providers*, and only once a request is concrete enough to ask about. The 

6rules in this module are the ones that are plain facts on the records the 

7catalogue publishes — the specialty's age window, the specialty's referral 

8flag, the plan's covered specialties, sites and providers — so they can be read 

9off the patient and the catalogue without guessing anything: 

10 

11- a child's age against ``SpecialtyRecord.min_age_months`` / ``max_age_months`` 

12- ``SpecialtyRecord.referral_required`` against ``PatientRecord.referrals`` 

13- ``InsurancePlanRecord.specialty_ids`` / ``location_ids`` / ``provider_ids`` 

14- ``ProviderRecord.insurer_ids_refused`` and ``ProviderRecord.leave`` 

15 

16Each returns the one ``DeclineReason`` that names it, which is why the closed 

17vocabulary's first eleven values map one-for-one onto the clinic's 

18restrictions: there is always a value for the rule that actually bit. 

19 

20The five insurance refusal shapes of problem 6 are not interchangeable, and 

21they do not all come from the same place: 

22 

23=========================== ============================== ================ 

24plan refuses the specialty ``specialty_not_covered`` catalogue, here 

25plan refuses the site ``location_not_covered`` catalogue, here 

26provider refuses the plan ``provider_not_in_network`` catalogue, here 

27plan demands its own ``insurer_referral_required`` ``blocked`` only 

28plan out of visits this year ``allowance_exhausted`` ``blocked`` only 

29=========================== ============================== ================ 

30 

31The last two are rules ``/clinic`` has no field for (``ClinicPlanResponse`` 

32carries no referral flag and no allowance), so nothing here derives them: the 

33restriction id ``/availability`` names is the reason, read as-is by 

34``rules.tools.check_eligibility``. Deriving them from a count of this year's 

35appointments would be a guess, and a guessed reason fails the case. 

36 

37The order the checks run in is the order the clinic would hit them, widest 

38first: an under-14 asking for general practice is an age refusal whatever their 

39insurer says. 

40"""

73def age_in_months(born: date, on: date) -> int: 

74 """Complete months lived. The 14th birthday is 168 months, to the day.""" 

75 months = (on.year - born.year) * 12 + (on.month - born.month) 

76 if on.day < born.day: 

77 months -= 1 

78 return months

81def specialty_for_age(catalogue: Catalogue, months: int) -> SpecialtyRecord | None: 

82 """The age-limited specialty whose window holds this age. 

83 

84 Paediatrics and general practice split at the 14th birthday with no gap and 

85 no overlap, so exactly one of them answers for a general complaint. This is 

86 what an age refusal redirects to. 

87 """ 

88 for specialty in catalogue.specialties: 

89 if specialty.min_age_months is None and specialty.max_age_months is None: 

90 continue 

91 if _age_fits(specialty, months): 

92 return specialty 

93 return None

96def _age_fits(specialty: SpecialtyRecord, months: int) -> bool: 

97 if specialty.min_age_months is not None and months < specialty.min_age_months: 

98 return False 

99 return not (specialty.max_age_months is not None and months > specialty.max_age_months)

102def resolve_plan( 

103 catalogue: Catalogue, patient: PatientRecord | None, named: str | None 

104) -> InsurancePlanRecord | None: 

105 """The plan this request is quoted against. 

106 

107 The caller's word wins: the second policy of problem 17 exists nowhere in 

108 the API, so a plan named on the call is the only way to find it. The one 

109 exception is self-pay, which is a plan a patient holds or does not. Quoting 

110 ``privado`` for someone whose record says otherwise is inventing cover to 

111 get past a refusal, so it falls back to the plan on the record. 

112 """ 

113 wanted = (named or "").strip().lower() 

114 on_file = (patient.insurer if patient else "").strip().lower() 

115 if wanted == SELF_PAY and on_file and on_file != SELF_PAY: 

116 wanted = on_file 

117 if not wanted: 

118 wanted = on_file 

119 if not wanted: 

120 return None 

121 return next((p for p in catalogue.insurance_plans if p.insurer_id.lower() == wanted), None)

124def holds_referral(patient: PatientRecord | None, specialty_id: str | None) -> bool: 

125 if patient is None or specialty_id is None: 

126 return False 

127 return specialty_id in {r.lower() for r in patient.referrals}

130def providers_for( 

131 catalogue: Catalogue, 

132 specialty_id: str | None, 

133 *, 

134 location_id: str | None = None, 

135 plan: InsurancePlanRecord | None = None, 

136 exclude: set[str] | None = None, 

137) -> list[ProviderRecord]: 

138 """Everyone who could take this request. The redirect list, in one place.""" 

139 out = [] 

140 for provider in catalogue.providers: 

141 if specialty_id and provider.specialty_id != specialty_id: 

142 continue 

143 if location_id and location_id not in provider.location_ids: 

144 continue 

145 if exclude and provider.provider_id in exclude: 

146 continue 

147 if plan and not accepts_plan(provider, plan): 

148 continue 

149 out.append(provider) 

150 return out

153def accepts_plan(provider: ProviderRecord, plan: InsurancePlanRecord) -> bool: 

154 """Whether this provider bills this plan. Refusal is per provider.""" 

155 insurer = plan.insurer_id.lower() 

156 if insurer in {i.lower() for i in provider.insurer_ids_refused}: 

157 return False 

158 # A plan that lists the providers it works with excludes everyone else. 

159 return not (plan.provider_ids and provider.provider_id not in plan.provider_ids)

162def sites_serving( 

163 catalogue: Catalogue, specialty_id: str | None, location_id: str | None 

164) -> list[LocationRecord]: 

165 """The sites where this request could physically happen.""" 

166 hosts = { 

167 loc_id 

168 for provider in catalogue.providers 

169 if not specialty_id or provider.specialty_id == specialty_id 

170 for loc_id in provider.location_ids 

171 } 

172 return [ 

173 loc 

174 for loc in catalogue.locations 

175 if loc.location_id in hosts and (not location_id or loc.location_id == location_id) 

176 ]

179def check_patient_rules( 

180 catalogue: Catalogue, 

181 patient: PatientRecord | None, 

182 *, 

183 specialty_id: str | None, 

184 location_id: str | None, 

185 plan: InsurancePlanRecord | None, 

186 today: date, 

187) -> RuleVerdict | None: 

188 """The rules that bite before any particular doctor is chosen. 

189 

190 ``None`` means these rules allow it — not that a slot exists, and not that 

191 the plan's own referral or allowance rules do, which only ``/availability`` 

192 knows. 

193 """ 

194 specialty = next((s for s in catalogue.specialties if s.specialty_id == specialty_id), None) 

195 

196 # 1. Age. The widest rule: it decides which specialty the caller belongs in 

197 # before anything about insurance is asked. 

198 if specialty and patient and patient.date_of_birth: 

199 months = age_in_months(patient.date_of_birth, today) 

200 if not _age_fits(specialty, months): 

201 correct = specialty_for_age(catalogue, months) 

202 return RuleVerdict( 

203 reason="not_eligible_age", 

204 detail=( 

205 f"{specialty.name} takes " 

206 f"{_window_text(specialty)}; the patient is {months} months old" 

207 ), 

208 redirect_to=( 

209 providers_for(catalogue, correct.specialty_id, location_id=location_id) 

210 if correct 

211 else [] 

212 ), 

213 ) 

214 

215 # 2. The specialty's own referral requirement, against the referrals the 

216 # directory record carries. No record is not the same as no referral: 

217 # without one this stands down and /availability answers, exactly as the 

218 # age rule above does. (``/directory`` has no lookup by id, so a record 

219 # is not always in hand.) 

220 if ( 

221 specialty 

222 and specialty.referral_required 

223 and patient 

224 and not holds_referral(patient, specialty_id) 

225 ): 

226 return RuleVerdict( 

227 reason="referral_required", 

228 detail=f"{specialty.name} needs a referral and the record holds none", 

229 ) 

230 

231 if plan is None: 

232 return None 

233 

234 # 3. The plan refuses the specialty. Adeslas and gynaecology: there is one 

235 # gynaecologist, so there is nowhere to redirect to. 

236 if plan.specialty_ids and specialty_id and specialty_id not in plan.specialty_ids: 

237 return RuleVerdict( 

238 reason="specialty_not_covered", 

239 detail=f"{plan.name} does not cover {specialty_id}", 

240 ) 

241 

242 # 4. The plan refuses the site. A refusal only when *no* site that could 

243 # serve the request is covered — ASISA covers physiotherapy at Centro 

244 # and Norte, and the one physiotherapist sits at Sur. 

245 if plan.location_ids: 

246 candidates = sites_serving(catalogue, specialty_id, location_id) 

247 if candidates and not any(loc.location_id in plan.location_ids for loc in candidates): 

248 named = f" at {location_id}" if location_id else "" 

249 return RuleVerdict( 

250 reason="location_not_covered", 

251 detail=f"{plan.name} does not cover {specialty_id}{named}", 

252 ) 

253 

254 # 5 and 6, the plan's own referral and its yearly allowance, are not here: 

255 # the catalogue publishes neither, and ``/availability`` names them in 

256 # ``blocked``. See the module docstring. 

257 return None

260def check_provider_rules( 

261 catalogue: Catalogue, 

262 provider: ProviderRecord | None, 

263 *, 

264 specialty_id: str | None, 

265 location_id: str | None, 

266 plan: InsurancePlanRecord | None, 

267 today: date, 

268) -> RuleVerdict | None: 

269 """The rules that bite once the caller has named a doctor. 

270 

271 Only consulted when ``/availability`` did not already name one: the live 

272 API is the authority on its own providers. 

273 """ 

274 if provider is None: 

275 return None 

276 specialty = specialty_id or provider.specialty_id 

277 

278 if plan and not accepts_plan(provider, plan): 

279 return RuleVerdict( 

280 reason="provider_not_in_network", 

281 detail=f"{provider.name} does not take {plan.name}", 

282 redirect_to=providers_for( 

283 catalogue, 

284 specialty, 

285 location_id=location_id, 

286 plan=plan, 

287 exclude={provider.provider_id}, 

288 ), 

289 ) 

290 

291 leave = next((lv for lv in provider.leave if lv.date_from <= today <= lv.date_to), None) 

292 if leave: 

293 return RuleVerdict( 

294 reason="provider_on_leave", 

295 detail=leave.reason or f"{provider.name} is on leave", 

296 redirect_to=providers_for( 

297 catalogue, 

298 specialty, 

299 location_id=location_id, 

300 plan=plan, 

301 exclude={provider.provider_id}, 

302 ), 

303 ) 

304 return None

307def _window_text(specialty: SpecialtyRecord) -> str: 

308 low, high = specialty.min_age_months, specialty.max_age_months 

309 if low is not None and high is not None: 

310 return f"{low}-{high} months" 

311 if low is not None: 

312 return f"{low} months and over" 

313 if high is not None: 

314 return f"up to {high} months" 

315 return "any age"
