1"""Process-wide settings, read once from the environment. 

2 

3Every value has a default that works with no key configured. When a key is 

4missing, the matching component runs in fake mode: 

5 

6- no ``PLATFORM_API_KEY`` -> fake clinic data and a dry-run submit client 

7- no voice keys -> the stub voice pipeline (beeps, no STT/LLM/TTS) 

8""" 

9 

10from __future__ import annotations 

11 

12import os 

13from dataclasses import dataclass, field 

14from pathlib import Path 

15 

16from dotenv import load_dotenv 

17 

18REPO_ROOT = Path(__file__).resolve().parent.parent 

19 

20load_dotenv(REPO_ROOT / ".env") 

21 

22 

23def _env(name: str, default: str = "") -> str: 

24 return os.environ.get(name, default).strip() 

25 

26 

27@dataclass(frozen=True) 

28class Settings: 

29 # Platform (the organisers' API: clinic reads + submit routes) 

30 platform_api_key: str = field(default_factory=lambda: _env("PLATFORM_API_KEY")) 

31 platform_api_base_url: str = field( 

32 default_factory=lambda: _env("PLATFORM_API_BASE_URL", "http://localhost:9999") 

33 ) 

34 

35 # Voice pipeline providers 

36 deepgram_api_key: str = field(default_factory=lambda: _env("DEEPGRAM_API_KEY")) 

37 openai_api_key: str = field(default_factory=lambda: _env("OPENAI_API_KEY")) 

38 openai_llm_model: str = field(default_factory=lambda: _env("OPENAI_LLM_MODEL", "gpt-4.1")) 

39 openai_tts_model: str = field( 

40 default_factory=lambda: _env("OPENAI_TTS_MODEL", "gpt-4o-mini-tts") 

41 ) 

42 openai_tts_voice: str = field(default_factory=lambda: _env("OPENAI_TTS_VOICE", "coral")) 

43 deepgram_stt_model: str = field(default_factory=lambda: _env("DEEPGRAM_STT_MODEL", "nova-3")) 

44 

45 # Server 

46 host: str = field(default_factory=lambda: _env("VORTEX_HOST", "0.0.0.0")) 

47 port: int = field(default_factory=lambda: int(_env("VORTEX_PORT", "7860"))) 

48 ws_path: str = field(default_factory=lambda: _env("VORTEX_WS_PATH", "/ws")) 

49 

50 # Forced modes. "auto" derives the mode from the keys above. 

51 # VORTEX_VOICE_MODE: auto | stub | pipecat 

52 voice_mode: str = field(default_factory=lambda: _env("VORTEX_VOICE_MODE", "auto")) 

53 # VORTEX_CLINIC_MODE: auto | fake | live 

54 clinic_mode: str = field(default_factory=lambda: _env("VORTEX_CLINIC_MODE", "auto")) 

55 

56 # Observability 

57 calls_log_path: Path = field( 

58 default_factory=lambda: Path(_env("VORTEX_CALLS_LOG", str(REPO_ROOT / "logs/calls.jsonl"))) 

59 ) 

60 

61 # Submission window: the platform closes it 30 s after the socket closes. 

62 # We keep a margin so a late retry still lands inside it. 

63 submit_window_secs: float = 30.0 

64 submit_deadline_margin_secs: float = 5.0 

65 

66 @property 

67 def clinic_is_live(self) -> bool: 

68 if self.clinic_mode == "live": 

69 return True 

70 if self.clinic_mode == "fake": 

71 return False 

72 return bool(self.platform_api_key) 

73 

74 @property 

75 def voice_is_pipecat(self) -> bool: 

76 if self.voice_mode == "pipecat": 

77 return True 

78 if self.voice_mode == "stub": 

79 return False 

80 return bool(self.deepgram_api_key and self.openai_api_key) 

81 

82 def describe(self) -> dict[str, object]: 

83 """A safe summary for logs and /health. Never includes key values.""" 

84 return { 

85 "clinic": "live" if self.clinic_is_live else "fake", 

86 "voice": "pipecat" if self.voice_is_pipecat else "stub", 

87 "platform_api_base_url": self.platform_api_base_url, 

88 "has_platform_key": bool(self.platform_api_key), 

89 "has_deepgram_key": bool(self.deepgram_api_key), 

90 "has_openai_key": bool(self.openai_api_key), 

91 "ws_path": self.ws_path, 

92 "calls_log_path": str(self.calls_log_path), 

93 } 

94 

95 

96_settings: Settings | None = None 

97 

98 

99def get_settings() -> Settings: 

100 global _settings 

101 if _settings is None: 

102 _settings = Settings() 

103 return _settings 

104 

105 

106def reset_settings() -> None: 

107 """Drop the cached settings. Tests call this after they change the environment.""" 

108 global _settings 

109 _settings = None
