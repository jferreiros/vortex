1"""Turn-taking and tool exposure. Consumed by ``line/pipecat_voice.py``. 

2 

3Owner: the conversation lane. 

4 

5Interruption handling is entirely ours (the platform does no barge-in). 

6 

7**Where interruptions live in pipecat 1.11.** ``PipelineParams`` no longer has 

8``allow_interruptions``: passing it is accepted and silently ignored. The 

9switch now belongs to the *user turn strategies* the user aggregator runs 

10(``pipecat.turns``). Two paths, chosen by ``soniox_turn_detection``: 

11 

12- ``True`` (default): Soniox's own endpoint detection ends the turn. The STT 

13 service is built with ``vad_force_turn_endpoint=False`` and 

14 ``should_interrupt=enable_interruptions``; it then installs 

15 ``ExternalUserTurnStrategies(enable_interruptions=...)`` on the aggregator 

16 itself through its metadata frame. Nothing else to wire. 

17- ``False``: pipecat's VAD starts and ends the turn. Then the aggregator must 

18 be given ``user_turn_strategies=user_turn_strategies(settings)`` (below), or 

19 it falls back to its defaults, which load the smart-turn v3 model and ignore 

20 ``enable_interruptions``. 

21 

22Either way the line lane passes the strategies, never a ``PipelineParams`` 

23flag. ``LLMUserAggregatorParams(user_turn_strategies=...)`` is the argument. 

24 

25Noise (problem 12) and the eight-second silence (problem 13) both live here: 

26the VAD thresholds keep a passing bus from becoming a barge-in, and 

27``user_idle_secs`` is when the aggregator fires ``on_user_turn_idle`` so the 

28agent can ask "are you still there?" (``prompt.idle_prompt_for``) instead of 

29letting the platform cut a quiet call. 

30"""

58@dataclass(frozen=True) 

59class TurnSettings: 

60 # The caller may talk over the agent while it reads options (problem 13). 

61 enable_interruptions: bool = True 

62 # In VAD mode, words the caller must say before a barge-in counts. One 

63 # word is "uh-huh" or the television; two is a correction. 

64 interrupt_min_words: int = 2 

65 # Noisy-caller settings (problem 12): a higher bar before Silero calls it 

66 # speech, so a bus going past does not become a barge-in. No denoiser in 

67 # front of STT: Deepgram/AssemblyAI both document worse WER after 

68 # suppression, and Soniox v5 is trained for telephony noise. 

69 vad_confidence: float = 0.85 

70 vad_start_secs: float = 0.3 

71 vad_stop_secs: float = 0.4 

72 vad_min_volume: float = 0.7 

73 # In VAD mode, seconds of silence after speech before the turn is over. 

74 # Longer than the mid-id pause ("one two, three four ... five six"). 

75 user_speech_timeout_secs: float = 1.2 

76 # Seconds of caller silence before the agent prompts again. 0 disables. 

77 # The difficult caller goes quiet for about eight seconds; the platform's 

78 # own cut-off for a quiet line is unpublished, so nudge before it could. 

79 user_idle_secs: float = 6.0 

80 exposed_tools: list[str] = field(default_factory=lambda: list(DEFAULT_EXPOSED_TOOLS)) 

81 

82 # --- Soniox STT --------------------------------------------------------- 

83 # Hints, not a lock: stt-rt-v5 still transcribes anything it hears, and with 

84 # language identification on it tags every token with the language it heard. 

85 # English first: it is the clinic's default and 69 of 73 public cases. 

86 stt_language_hints: tuple[str, ...] = ("en", "es", "ca") 

87 # True -> Soniox's own endpoint detection ends the turn (vad_force_turn_endpoint=False) 

88 # False -> pipecat's VAD ends the turn and finalises Soniox 

89 soniox_turn_detection: bool = True 

90 # The three below only bite when soniox_turn_detection is True. 

91 # 1500 ms tolerates the pause callers make mid-DNI ("twelve, thirty-four ..."). 

92 stt_max_endpoint_delay_ms: int = 1500 

93 stt_endpoint_sensitivity: float = 0.3 

94 stt_endpoint_latency_adjustment_level: int = 2 

95

98 return TurnSettings()

101def user_turn_strategies(settings: TurnSettings | None = None) -> Any | None: 

102 """The pipecat ``UserTurnStrategies`` for these settings, or ``None``. 

103 

104 ``None`` in Soniox turn-detection mode: the STT service installs 

105 ``ExternalUserTurnStrategies`` itself, and a value passed here would 

106 override it and break turn endings. In VAD mode it returns a VAD start 

107 strategy gated on ``interrupt_min_words`` while the bot speaks, and a 

108 speech-timeout stop strategy, both honouring ``enable_interruptions``. 

109 

110 Imports pipecat lazily so the server and the tests start without it. 

111 """ 

112 turns = settings or default_turn_settings() 

113 if turns.soniox_turn_detection: 

114 return None 

115 from pipecat.turns.user_start import ( 

116 MinWordsUserTurnStartStrategy, 

117 VADUserTurnStartStrategy, 

118 ) 

119 from pipecat.turns.user_stop import SpeechTimeoutUserTurnStopStrategy 

120 from pipecat.turns.user_turn_strategies import UserTurnStrategies 

121 

122 start: list[Any] = [ 

123 VADUserTurnStartStrategy(enable_interruptions=turns.enable_interruptions), 

124 ] 

125 if turns.interrupt_min_words > 1: 

126 start.append( 

127 MinWordsUserTurnStartStrategy( 

128 min_words=turns.interrupt_min_words, 

129 enable_interruptions=turns.enable_interruptions, 

130 ) 

131 ) 

132 stop = [SpeechTimeoutUserTurnStopStrategy(user_speech_timeout=turns.user_speech_timeout_secs)] 

133 return UserTurnStrategies(start=start, stop=stop)
