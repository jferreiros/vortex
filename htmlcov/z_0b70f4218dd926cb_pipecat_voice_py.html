1"""The real voice pipeline: pipecat over the Twilio-shaped socket. 

2 

3 transport.input -> STT (Soniox stt-rt-v5) -> language watcher 

4 -> user aggregator -> LLM (OpenAI-compatible, EU) 

5 -> TTS (Google Cloud, Azure Neural or Deepgram Aura-2) 

6 -> transport.output -> assistant aggregator 

7 

8Soniox transcribes, an OpenAI-compatible endpoint named by ``LLM_PROVIDER`` 

9answers, and the voice is Google Cloud Text-to-Speech — the only provider here 

10with Catalan, Galician *and* Basque. 

11 

12Two TTS services can run at once. ``VORTEX_TTS_PROVIDER`` speaks Spanish and 

13``VORTEX_TTS_PROVIDER_ALT`` speaks what the primary cannot, so 

14``VORTEX_TTS_PROVIDER=elevenlabs`` with the default ``_ALT=google`` gives 

15ElevenLabs Spanish and Google ca/gl/eu. When the two are the same (the 

16default, google/google) there is one service and one voice-swap path. 

17 

18One pipeline per socket. The serializer takes the ``stream_sid`` of this call, 

19the context takes this call's prompt, and every tool handler closes over this 

20call's ``ToolContext``. Nothing here is module-level state. 

21 

22Owner: the line lane for the wiring; the conversation lane for prompt, turn 

23settings, STT vocabulary and language (it edits ``vortex/conversation/*``). 

24 

25Imports of pipecat live inside the function so the server starts, and the 

26smoke test runs, with the stub pipeline when the keys are missing. 

27 

28TODO(line): 

29- Run one real call end to end once the four keys exist. 

30- Confirm the 8 kHz µ-law path: serializer ``twilio_sample_rate=8000`` in, and 

31 the TTS asked for 8 kHz PCM out (Google LINEAR16 @ 8000, ElevenLabs 

32 ``pcm_8000``). Check for choppy audio. 

33- Decide the idle policy: the platform cuts a call that goes quiet. Keep a 

34 user-idle prompt so the agent is never silent for long. 

35- Hang up from our side when the agent says goodbye (send EndFrame, let the 

36 platform close the socket). auto_hang_up stays False: no Twilio account. 

37"""

101 ctx.log.event("voice.mode", mode="pipecat", **_providers(settings))

130 ),

132 vad_force_turn_endpoint=not turns.soniox_turn_detection,

134 )

154 # ---- tools: every registry entry becomes a function the model can call ---- 

155 def make_handler(tool_name: str): 

156 async def handler(params: FunctionCallParams) -> None: 

157 try: 

158 # Through the session, not the registry: it remembers what the 

159 # end-of-call fallback needs (the last refusal, the last 

160 # prepared action) and records the model's own submissions. 

161 result = await session.call_tool(tool_name, dict(params.arguments)) 

162 await params.result_callback(result.model_dump(mode="json")) 

163 except Exception as exc: # the model must hear about failures, typed 

164 await params.result_callback({"error": f"{type(exc).__name__}: {exc}"}) 

165 

166 return handler 

167 

168 schemas: list[Any] = [] 

169 for fn in registry.function_schemas(turns.exposed_tools): 

170 params_schema = fn["parameters"] 

171 properties = dict(params_schema["properties"]) 

172 if params_schema.get("$defs"): 

173 # Nested models (Slot, Action ...) need their definitions inline. 

174 properties["$defs"] = params_schema["$defs"] 

175 schemas.append( 

176 FunctionSchema( 

177 name=fn["name"], 

178 description=fn["description"], 

179 properties=properties, 

180 required=params_schema["required"], 

181 ) 

182 ) 

183 llm.register_function(fn["name"], make_handler(fn["name"])) 

184 

185 context = LLMContext(initial_messages(ctx.now), tools=ToolsSchema(standard_tools=schemas)) 

186 aggregators = LLMContextAggregatorPair( 

187 context, 

188 user_params=LLMUserAggregatorParams( 

189 vad_analyzer=SileroVADAnalyzer( 

190 params=VADParams( 

191 confidence=turns.vad_confidence, 

192 start_secs=turns.vad_start_secs, 

193 stop_secs=turns.vad_stop_secs, 

194 min_volume=turns.vad_min_volume, 

195 ) 

196 ), 

197 user_idle_timeout=turns.user_idle_secs, 

198 ), 

199 ) 

200 

201 stages: list[Any] = [transport.input(), stt] 

202 if settings.tts_supports_language_switch: 

203 # Only worth a processor when the pair can say more than one language. 

204 # ElevenLabs alone is Spanish-only, so the watcher would be a no-op. 

205 stages.append(_LanguageWatcher(session, language_state)) 

206 stages += [ 

207 aggregators.user(), 

208 llm, 

209 tts, 

210 transport.output(), 

211 aggregators.assistant(), 

212 ] 

213 

214 task = PipelineTask( 

215 Pipeline(stages), 

216 params=PipelineParams( 

217 audio_in_sample_rate=LINE_SAMPLE_RATE, 

218 audio_out_sample_rate=LINE_SAMPLE_RATE, 

219 enable_metrics=True, 

220 ), 

221 observers=[_CallLogObserver(session)], 

222 ) 

223 

224 @transport.event_handler("on_client_connected") 

225 async def _on_connected(transport: Any, client: Any) -> None: 

226 ctx.log.assistant_turn(GREETING) 

227 await task.queue_frames([TTSSpeakFrame(GREETING)]) 

228 

229 @transport.event_handler("on_client_disconnected") 

230 async def _on_disconnected(transport: Any, client: Any) -> None: 

231 ctx.log.event("voice.client_disconnected") 

232 await task.cancel() 

233 

234 @aggregators.user().event_handler("on_user_turn_idle") 

235 async def _on_user_idle(aggregator: Any, *args: Any) -> None: 

236 # TODO(conversation): the "are you still there?" prompt lives here. 

237 ctx.log.event("voice.user_idle") 

238 

239 runner = PipelineRunner(handle_sigint=False) 

240 await runner.run(task) 

241 return "pipeline_finished"

253class _LanguageState: 

254 """The language this call is being spoken in, right now. 

255 

256 One instance per call. The watcher writes it as the caller switches; the 

257 router reads it to decide which TTS branch the next text frame belongs to. 

258 Frames cross the pipeline in order, so a write from the watcher is always 

259 visible to the router by the time the matching text arrives. 

260 """ 

261 

262 __slots__ = ("language",) 

263 

264 def __init__(self, language: str = DEFAULT_LANGUAGE) -> None: 

265 self.language = language

276def _llm_extra_body(settings: Any) -> dict[str, Any]: 

277 """Request fields that turn reasoning off, for every host we might hit. 

278 

279 vLLM/SGLang read ``chat_template_kwargs.enable_thinking``; Helmcode reads 

280 ``reasoning_effort`` ("none" skips the phase on qwen3.6/gemma4). Hosts 

281 ignore the one they do not know. 

282 """ 

283 # pipecat spreads this dict as keyword arguments of the SDK's 

284 # ``chat.completions.create``: ``reasoning_effort`` is one of its 

285 # parameters, ``chat_template_kwargs`` is not and has to travel in 

286 # ``extra_body`` to reach the request JSON. 

287 extra: dict[str, Any] = {} 

288 if settings.llm_disable_thinking: 

289 extra["extra_body"] = {"chat_template_kwargs": {"enable_thinking": False}} 

290 if settings.llm_reasoning_effort: 

291 extra["reasoning_effort"] = settings.llm_reasoning_effort 

292 return extra

295def _make_tts(settings: Any, provider: str | None = None) -> Any: 

296 """Build one TTS service, asked for 8 kHz PCM. 

297 

298 Google encodes LINEAR16 at 8000; ElevenLabs maps the same rate to its 

299 ``pcm_8000`` output format. The Twilio serializer does the µ-law companding 

300 on the way out either way, so the wire format never changes with the 

301 provider. 

302 """ 

303 name = provider or settings.tts_provider 

304 voice, language = tts_voice_for(DEFAULT_LANGUAGE, settings, name) 

305 if not voice: 

306 # Builds fine, then fails on every utterance. Say so once, loudly. 

307 log.warning("TTS provider %s has no Spanish voice configured", name) 

308 

309 if name == "elevenlabs": 

310 from pipecat.services.elevenlabs.tts import ElevenLabsTTSService 

311 

312 return ElevenLabsTTSService( 

313 api_key=settings.elevenlabs_api_key, 

314 # A WebSocket origin, so an AI Gateway in front of ElevenLabs goes 

315 # here. Empty keeps the service's own default. 

316 **({"url": settings.elevenlabs_base_url} if settings.elevenlabs_base_url else {}), 

317 sample_rate=LINE_SAMPLE_RATE, 

318 settings=ElevenLabsTTSService.Settings( 

319 voice=voice, 

320 model=settings.elevenlabs_model, 

321 language=language, 

322 ), 

323 ) 

324 

325 # The HTTP service, not the streaming ``GoogleTTSService``: streaming only 

326 # speaks Chirp 3 HD / Journey, and ca/gl/eu exist solely as Standard 

327 # voices. The HTTP one takes both families, so a single service covers 

328 # every language through a voice swap. 

329 from pipecat.services.google.tts import GoogleHttpTTSService 

330 

331 return GoogleHttpTTSService( 

332 # Inline JSON wins when both are set, matching pipecat's own order. 

333 credentials=settings.google_tts_credentials_json or None, 

334 credentials_path=settings.google_application_credentials or None, 

335 sample_rate=LINE_SAMPLE_RATE, 

336 settings=GoogleHttpTTSService.Settings(voice=voice, language=language), 

337 )

340def _TTSRouter( # noqa: N802 - factory that returns a processor 

341 settings: Any, state: _LanguageState, primary: Any, alternate: Any 

342) -> Any: 

343 """Route each spoken language to the service that can say it. 

344 

345 A ``ParallelPipeline`` with two branches, each fronted by a 

346 ``FunctionFilter`` keyed on ``state.language``: the primary branch takes 

347 every language in its capability set, the alternate branch takes the rest. 

348 Only one branch is ever fed text, so only one branch produces audio. 

349 

350 Why the filters are shaped this way: 

351 

352 - ``FunctionFilter`` lets ``StartFrame``/``EndFrame``/``CancelFrame`` 

353 through unconditionally, so both services start, stop and cancel with the 

354 pipeline even while idle. 

355 - System frames are left unfiltered (``filter_system_frames`` off), so an 

356 interruption reaches both services and neither is left mid-utterance. 

357 ``ParallelPipeline`` de-duplicates by frame id on the way out, so a frame 

358 that crossed both branches still leaves once. 

359 - Everything that makes a TTS speak — ``TextFrame``, ``TTSSpeakFrame``, the 

360 ``LLMFullResponse*`` brackets — is a data or control frame, so it is 

361 gated, and the idle branch stays silent. 

362 - ``TTSUpdateSettingsFrame`` is gated too, which is what makes the watcher 

363 work unchanged: it writes the language first, so its voice update lands 

364 on whichever branch is about to speak. 

365 

366 This is the same construction pipecat's own ``ServiceSwitcher`` uses 

367 (``ParallelPipeline`` of filter + service). We key the filters on the 

368 detected language directly instead of driving a switcher with 

369 ``ManuallySwitchServiceFrame``: routing is a pure function of the language, 

370 so a second source of truth about which service is "active" would only be 

371 something else to keep in sync. 

372 """ 

373 from pipecat.pipeline.parallel_pipeline import ParallelPipeline 

374 from pipecat.processors.filters.function_filter import FunctionFilter 

375 from pipecat.processors.frame_processor import FrameDirection 

376 

377 primary_languages = settings.tts_languages(settings.tts_provider) 

378 

379 async def to_primary(_frame: Any) -> bool: 

380 return state.language in primary_languages 

381 

382 async def to_alternate(_frame: Any) -> bool: 

383 return state.language not in primary_languages 

384 

385 def gate(fn: Any) -> Any: 

386 # enable_direct_mode: the predicate is a set lookup, not worth a task. 

387 return FunctionFilter( 

388 filter=fn, direction=FrameDirection.DOWNSTREAM, enable_direct_mode=True 

389 ) 

390 

391 return ParallelPipeline([gate(to_primary), primary], [gate(to_alternate), alternate])

394def _LanguageWatcher( # noqa: N802 - factory that returns a processor 

395 session: CallSession, state: _LanguageState | None = None 

396): 

397 """Switch the voice when the caller switches language. 

398 

399 Soniox tags each transcription frame with the language it heard. When that 

400 flips we write the new language into the shared ``_LanguageState`` and push 

401 a ``TTSUpdateSettingsFrame`` downstream; the TTS service applies the delta 

402 in place, so the voice changes without rebuilding the pipeline. There is no 

403 public ``update_settings()`` coroutine on ``TTSService`` in pipecat 1.11 — 

404 the control frame is the supported way in. 

405 

406 The voice comes from the provider that serves the new language, which is 

407 the primary unless the alternate is the one that covers it. The state is 

408 written *before* the frame is pushed, so when a router is in the pipeline 

409 the update travels down the branch that is about to speak. 

410 

411 ``TTSService._update_settings`` converts the pipecat ``Language`` we send 

412 into the provider's own code before storing it, which is what 

413 ``GoogleHttpTTSService.run_tts`` passes as ``language_code`` next to the 

414 new voice name. Google's verified map only lists the Chirp 3 HD locales, 

415 so ca/gl/eu log a "not verified" warning and fall through to the full code 

416 ("ca-ES", "gl-ES", "eu-ES") — the right value either way. 

417 """ 

418 from pipecat.frames.frames import Frame, TranscriptionFrame, TTSUpdateSettingsFrame 

419 from pipecat.processors.frame_processor import FrameDirection, FrameProcessor 

420 from pipecat.services.settings import TTSSettings 

421 

422 language_state = state if state is not None else _LanguageState() 

423 

424 class LanguageWatcher(FrameProcessor): 

425 def __init__(self) -> None: 

426 super().__init__() 

427 self._state = language_state 

428 

429 async def process_frame(self, frame: Frame, direction: FrameDirection) -> None: 

430 await super().process_frame(frame, direction) 

431 if isinstance(frame, TranscriptionFrame): 

432 await self._maybe_switch(frame) 

433 await self.push_frame(frame, direction) 

434 

435 async def _maybe_switch(self, frame: TranscriptionFrame) -> None: 

436 try: 

437 settings = session.settings 

438 language = detect_language(frame.text, hint=frame.language) 

439 if language == self._state.language: 

440 return 

441 provider = settings.tts_provider_for(language) 

442 voice, tts_language = tts_voice_for(language, settings, provider) 

443 previous, self._state.language = self._state.language, language 

444 session.ctx.log.event( 

445 "voice.language_switch", 

446 was=previous, 

447 now=language, 

448 voice=voice, 

449 provider=provider, 

450 ) 

451 await self.push_frame( 

452 TTSUpdateSettingsFrame(delta=TTSSettings(voice=voice, language=tts_language)), 

453 FrameDirection.DOWNSTREAM, 

454 ) 

455 except Exception as exc: # a failed switch must never end the call 

456 log.warning("language switch failed: %s", exc) 

457 

458 return LanguageWatcher()

461def _CallLogObserver(session: CallSession): # noqa: N802 - factory that returns an observer 

462 """Log user and assistant text, and count media frames, from the frame stream.""" 

463 from pipecat.frames.frames import ( 

464 InputAudioRawFrame, 

465 OutputAudioRawFrame, 

466 TranscriptionFrame, 

467 TTSTextFrame, 

468 ) 

469 from pipecat.observers.base_observer import BaseObserver, FramePushed 

470 

471 class Observer(BaseObserver): 

472 async def on_push_frame(self, data: FramePushed) -> None: 

473 frame = data.frame 

474 if isinstance(frame, TranscriptionFrame): 

475 session.ctx.log.user_turn(frame.text) 

476 elif isinstance(frame, TTSTextFrame): 

477 session.ctx.log.assistant_turn(frame.text) 

478 elif isinstance(frame, InputAudioRawFrame): 

479 session.media_frames_in += 1 

480 elif isinstance(frame, OutputAudioRawFrame): 

481 session.media_frames_out += 1 

482 

483 return Observer()
